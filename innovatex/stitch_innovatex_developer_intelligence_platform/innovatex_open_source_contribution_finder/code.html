// Pre-compiled RFC 7230 header validation token
const HEADER_KEY_REGEX = /^[a-zA-Z0-9\-_]+$/;

export class NextResponse extends Response {
  /**
   * Validates custom headers against HTTP standards before dispatch
   */
  public validateHeader(key: string, value: string): boolean {
    if (!HEADER_KEY_REGEX.test(key)) {
      if (process.env.NODE_ENV !== 'production') {
        console.warn(`[NextResponse] Discarding invalid header key "${key}".`);
      }
      return false;
    }
    return true;
  }
